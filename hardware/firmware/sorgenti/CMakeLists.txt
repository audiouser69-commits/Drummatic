cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(drummatic_ctrl C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(drummatic_ctrl main.c usb_descriptors.c)
pico_generate_pio_header(drummatic_ctrl ${CMAKE_CURRENT_LIST_DIR}/ws2812.pio)
target_include_directories(drummatic_ctrl PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(drummatic_ctrl PRIVATE pico_stdlib pico_unique_id hardware_pio hardware_adc tinyusb_device)
pico_add_extra_outputs(drummatic_ctrl)
